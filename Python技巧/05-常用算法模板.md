# Python 常用算法模板

## 一、快速幂

```python
def quick_pow(a, b, mod=10**9+7):
    res = 1
    while b:
        if b & 1:
            res = res * a % mod
        a = a * a % mod
        b >>= 1
    return res
```

## 二、最大公约数

```python
def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

# 或使用 math 模块
import math
g = math.gcd(12, 18)  # 6
```

## 三、最小公倍数

```python
def lcm(a, b):
    return a // gcd(a, b) * b

# 或使用 math 模块（Python 3.9+）
import math
l = math.lcm(12, 18)  # 36
```

## 四、素数筛

```python
def sieve(n):
    is_prime = [True] * (n + 1)
    is_prime[0] = is_prime[1] = False
    for i in range(2, int(n**0.5) + 1):
        if is_prime[i]:
            for j in range(i * i, n + 1, i):
                is_prime[j] = False
    return is_prime
```

## 五、前缀和

```python
def prefix_sum(arr):
    n = len(arr)
    prefix = [0] * (n + 1)
    for i in range(1, n + 1):
        prefix[i] = prefix[i - 1] + arr[i - 1]
    return prefix

# 区间 [l, r] 的和（0-based）
def range_sum(prefix, l, r):
    return prefix[r + 1] - prefix[l]
```

## 六、差分

```python
def difference(arr, queries):
    n = len(arr)
    diff = [0] * (n + 2)
    
    for l, r, val in queries:
        diff[l] += val
        diff[r + 1] -= val
    
    # 还原
    current = 0
    result = []
    for i in range(n):
        current += diff[i]
        result.append(current)
    
    return result
```

## 七、二分查找

```python
def binary_search(arr, target):
    left, right = 0, len(arr) - 1
    while left <= right:
        mid = left + (right - left) // 2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    return -1

# 二分答案
def binary_answer(left, right, check):
    ans = 0
    while left <= right:
        mid = left + (right - left) // 2
        if check(mid):
            ans = mid
            left = mid + 1
        else:
            right = mid - 1
    return ans
```

## 八、DFS（深度优先搜索）

```python
def dfs(node, visited, graph):
    visited.add(node)
    print(node)
    
    for neighbor in graph[node]:
        if neighbor not in visited:
            dfs(neighbor, visited, graph)

# 使用
graph = {
    0: [1, 2],
    1: [0, 3],
    2: [0, 3],
    3: [1, 2]
}
visited = set()
dfs(0, visited, graph)
```

## 九、BFS（广度优先搜索）

```python
from collections import deque

def bfs(start, visited, graph):
    queue = deque([start])
    visited.add(start)
    
    while queue:
        node = queue.popleft()
        print(node)
        
        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)

# 使用
visited = set()
bfs(0, visited, graph)
```
