# Python 输入输出

## 一、基本输入

```python
# 单个输入
n = int(input())

# 多个输入
a, b = map(int, input().split())
a, b = map(int, input().split(','))

# 列表输入
arr = list(map(int, input().split()))
arr = [int(x) for x in input().split()]

# 字符串输入
s = input()
s = input().strip()     # 去除首尾空白
```

## 二、多组输入

```python
# 方式1：EOF 结束
while True:
    try:
        a, b = map(int, input().split())
        print(a + b)
    except EOFError:
        break

# 方式2：指定组数
t = int(input())
for _ in range(t):
    a, b = map(int, input().split())
    print(a + b)

# 方式3：特殊值结束
while True:
    a, b = map(int, input().split())
    if a == 0 and b == 0:
        break
    print(a + b)
```

## 三、快速输入

```python
import sys

# 读取所有输入
data = sys.stdin.read().split()
n = int(data[0])
arr = list(map(int, data[1:]))

# 逐行读取
for line in sys.stdin:
    a, b = map(int, line.split())
    print(a + b)
```

## 四、基本输出

```python
# 普通输出
print("hello")
print(123)

# 格式化输出
print(f"姓名: {name}, 年龄: {age}")
print("姓名: {}, 年龄: {}".format(name, age))

# 不换行
print("hello", end=" ")
print("world")  # "hello world"

# 多个输出
print(a, b, c, sep=", ")  # 用逗号分隔
```

## 五、快速输出

```python
import sys

# 方式1：使用 write
sys.stdout.write("hello\n")
sys.stdout.write(str(123) + "\n")

# 方式2：收集输出
output = []
for i in range(5):
    output.append(str(i))
sys.stdout.write("\n".join(output))
```

## 六、文件输入输出

```python
# 读取文件
with open("input.txt", "r") as f:
    content = f.read()      # 全部读取
    lines = f.readlines()   # 按行读取
    line = f.readline()     # 读取一行

# 写入文件
with open("output.txt", "w") as f:
    f.write("hello\n")
    f.writelines(lines)

# 追加写入
with open("output.txt", "a") as f:
    f.write("append\n")
```

## 七、调试输出

```python
# 打印变量值
print(f"n = {n}, arr = {arr}")

# 打印类型
print(type(n))
print(type(arr))

# 打印长度
print(len(arr))

# 打印列表索引和值
for i, x in enumerate(arr):
    print(f"arr[{i}] = {x}")
```
