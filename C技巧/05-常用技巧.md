# C 常用技巧

## 一、常用宏定义

```c
#define MAX(a, b) ((a) > (b) ? (a) : (b))
#define MIN(a, b) ((a) < (b) ? (a) : (b))
#define ABS(x) ((x) < 0 ? -(x) : (x))
#define SWAP(a, b) {typeof(a) temp = a; a = b; b = temp;}

// 循环宏
#define FOR(i, a, b) for (int i = (a); i < (b); i++)
#define ROF(i, a, b) for (int i = (a); i >= (b); i--)

// 调试宏
#define DEBUG(x) printf("%s = %d\n", #x, x)
```

## 三、结构体

```c
// 定义结构体
struct Point {
    int x;
    int y;
};

// 使用
struct Point p1;
p1.x = 10;
p1.y = 20;

// 结构体指针
struct Point *p = &p1;
p->x = 30;
p->y = 40;

// typedef 简化
typedef struct Point Point;
Point p2;
p2.x = 50;
p2.y = 60;
```

## 四、位运算技巧

```c
// 判断奇偶
if (n & 1) {  // 奇数
    printf("奇数\n");
} else {      // 偶数
    printf("偶数\n");
}

// 交换两个数（不使用临时变量）
a ^= b;
b ^= a;
a ^= b;

// 判断是否为2的幂
if ((n & (n - 1)) == 0) {
    printf("是2的幂\n");
}

// 获取最低位的1
int lowbit = n & (-n);

// 删除最低位的1
n &= (n - 1);

// 统计二进制中1的个数
int count_ones(int n) {
    int count = 0;
    while (n) {
        count += n & 1;
        n >>= 1;
    }
    return count;
}
```

## 六、时间测量

```c

// 你的代码
```

## 七、调试技巧

```c
// 条件编译调试
#ifdef DEBUG
    printf("调试信息: %d\n", value);
#endif

// 断言
#include <assert.h>
assert(n >= 0);  // 如果n<0，程序终止

// 打印数组
void print_array(int arr[], int n) {
    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

// 打印指针地址
int x = 10;
int *p = &x;
printf("x的地址: %p\n", (void*)&x);
printf("p的值: %p\n", (void*)p);
```
