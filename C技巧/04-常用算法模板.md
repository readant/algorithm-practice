# C 常用算法模板

## 一、快速幂

```c
long long quick_pow(long long a, long long b, long long mod) {
    long long res = 1;
    while (b) {
        if (b & 1) res = res * a % mod;
        a = a * a % mod;
        b >>= 1;
    }
    return res;
}
```

## 二、最大公约数

```c
int gcd(int a, int b) {
    while (b) {
        int temp = b;
        b = a % b;
        a = temp;
    }
    return a;
}
```

## 三、最小公倍数

```c
int lcm(int a, int b) {
    return a / gcd(a, b) * b;
}
```

## 四、素数筛

```c
void sieve(int n, int is_prime[]) {
    for (int i = 0; i <= n; i++) {
        is_prime[i] = 1;
    }
    is_prime[0] = is_prime[1] = 0;
    
    for (int i = 2; i * i <= n; i++) {
        if (is_prime[i]) {
            for (int j = i * i; j <= n; j += i) {
                is_prime[j] = 0;
            }
        }
    }
}
```

## 五、前缀和

```c
void prefix_sum(int arr[], int n, long long prefix[]) {
    prefix[0] = 0;
    for (int i = 1; i <= n; i++) {
        prefix[i] = prefix[i - 1] + arr[i - 1];
    }
}

// 区间 [l, r] 的和（0-based）
long long range_sum(long long prefix[], int l, int r) {
    return prefix[r + 1] - prefix[l];
}
```

## 六、差分

```c
void difference(int arr[], int n, int diff[]) {
    for (int i = 0; i <= n + 1; i++) {
        diff[i] = 0;
    }
}

// 区间 [l, r] 加 val
void add_range(int diff[], int l, int r, int val) {
    diff[l] += val;
    diff[r + 1] -= val;
}

// 还原
void restore(int diff[], int n, int result[]) {
    int current = 0;
    for (int i = 0; i < n; i++) {
        current += diff[i];
        result[i] = current;
    }
}
```

## 七、二分查找

```c
int binary_search(int arr[], int n, int target) {
    int left = 0, right = n - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (arr[mid] == target) {
            return mid;
        } else if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    return -1;
}

// 二分答案
typedef int (*CheckFunc)(int);

int binary_answer(int left, int right, CheckFunc check) {
    int ans = 0;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (check(mid)) {
            ans = mid;
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    return ans;
}
```

## 八、排序

```c
// 比较函数
int compare(const void *a, const void *b) {
    return (*(int*)a - *(int*)b);
}

// 使用 qsort
void sort_array(int arr[], int n) {
    qsort(arr, n, sizeof(int), compare);
}

// 冒泡排序
void bubble_sort(int arr[], int n) {
    for (int i = 0; i < n - 1; i++) {
        int swapped = 0;
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
                swapped = 1;
            }
        }
        if (!swapped) break;
    }
}

// 选择排序
void selection_sort(int arr[], int n) {
    for (int i = 0; i < n - 1; i++) {
        int min_idx = i;
        for (int j = i + 1; j < n; j++) {
            if (arr[j] < arr[min_idx]) {
                min_idx = j;
            }
        }
        int temp = arr[i];
        arr[i] = arr[min_idx];
        arr[min_idx] = temp;
    }
}
```
