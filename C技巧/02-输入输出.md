# C 输入输出

## 一、基本输入输出

```c
#include <stdio.h>

// 输出
printf("hello\n");
printf("数字: %d\n", 123);
printf("浮点数: %.2f\n", 3.14159);
printf("字符串: %s\n", "hello");

// 输入
int n;
scanf("%d", &n);

float f;
scanf("%f", &f);

char str[100];
scanf("%s", str);           // 读一个单词
```

## 二、格式化输出

```c
// 整数
printf("%d\n", 123);        // 123
printf("%5d\n", 123);       // "  123"（宽度5）
printf("%05d\n", 123);      // "00123"（补零）
printf("%-5d\n", 123);      // "123  "（左对齐）

// 浮点数
printf("%f\n", 3.14159);    // 3.141590
printf("%.2f\n", 3.14159);  // 3.14
printf("%10.2f\n", 3.14159);// "      3.14"

// 科学计数法
printf("%e\n", 3.14159);    // 3.141590e+00

// 十六进制
printf("%x\n", 255);        // ff
printf("%X\n", 255);        // FF
```

## 三、多组输入

```c
// 方式1：EOF 结束
int a, b;
while (scanf("%d %d", &a, &b) != EOF) {
    printf("%d\n", a + b);
}

// 方式2：指定组数
int t;
scanf("%d", &t);
while (t--) {
    int a, b;
    scanf("%d %d", &a, &b);
    printf("%d\n", a + b);
}

// 方式3：特殊值结束
while (scanf("%d %d", &a, &b) == 2 && a != 0 && b != 0) {
    printf("%d\n", a + b);
}
```

## 四、字符串输入

```c
char str[100];

// 读一个单词
scanf("%s", str);

// 读一行
gets(str);                  // 不推荐（不安全）
fgets(str, 100, stdin);     // 推荐

// 混合使用时的注意事项
int n;
scanf("%d", &n);
getchar();                  // 清空缓冲区
fgets(str, 100, stdin);
```

## 五、字符输入输出

```c
// 输入字符
char c;
scanf("%c", &c);
c = getchar();

// 输出字符
putchar(c);
printf("%c\n", c);

// 判断字符
if (isdigit(c)) {           // 数字
    printf("是数字\n");
}
if (isalpha(c)) {           // 字母
    printf("是字母\n");
}
if (islower(c)) {           // 小写
    printf("是小写\n");
}
if (isupper(c)) {           // 大写
    printf("是大写\n");
}
```

## 六、文件输入输出

```c
FILE *fp;

// 写文件
fp = fopen("output.txt", "w");
if (fp == NULL) {
    printf("文件打开失败\n");
    return 1;
}
fprintf(fp, "hello\n");
fclose(fp);

// 读文件
fp = fopen("input.txt", "r");
if (fp == NULL) {
    printf("文件打开失败\n");
    return 1;
}
char line[100];
while (fgets(line, 100, fp) != NULL) {
    printf("%s", line);
}
fclose(fp);

// 追加写入
fp = fopen("output.txt", "a");
fprintf(fp, "append\n");
fclose(fp);
```

## 七、快速输入输出

```c
// 快速读入整数
int read_int() {
    int x = 0, f = 1;
    char ch = getchar();
    while (ch < '0' || ch > '9') {
        if (ch == '-') f = -1;
        ch = getchar();
    }
    while (ch >= '0' && ch <= '9') {
        x = x * 10 + ch - '0';
        ch = getchar();
    }
    return x * f;
}

// 快速输出整数
void write_int(int x) {
    if (x < 0) {
        putchar('-');
        x = -x;
    }
    if (x > 9) {
        write_int(x / 10);
    }
    putchar(x % 10 + '0');
}
```
