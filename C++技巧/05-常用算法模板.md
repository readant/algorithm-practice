# C++ 常用算法模板

## 一、快速幂

```cpp
ll quick_pow(ll a, ll b, ll mod) {
    ll res = 1;
    while (b) {
        if (b & 1) res = res * a % mod;
        a = a * a % mod;
        b >>= 1;
    }
    return res;
}
```

## 二、最大公约数

```cpp
int gcd(int a, int b) {
    return b == 0 ? a : gcd(b, a % b);
}

// 或使用 STL
#include <numeric>
int g = std::gcd(a, b);  // C++17
```

## 三、最小公倍数

```cpp
int lcm(int a, int b) {
    return a / gcd(a, b) * b;
}

// 或使用 STL
#include <numeric>
int l = std::lcm(a, b);  // C++17
```

## 四、素数筛

```cpp
vector<bool> is_prime(n + 1, true);
is_prime[0] = is_prime[1] = false;

for (int i = 2; i * i <= n; ++i) {
    if (is_prime[i]) {
        for (int j = i * i; j <= n; j += i) {
            is_prime[j] = false;
        }
    }
}
```

vector<ll> prefix(n + 1, 0);

## 六、差分

```cpp
vector<int> diff(n + 2, 0);

// 区间 [l, r] 加 val
diff[l] += val;
diff[r + 1] -= val;

// 还原
int current = 0;
for (int i = 1; i <= n; ++i) {
    current += diff[i];
    arr[i] = current;
}
```

## 七、二分查找

```cpp
int binary_search(vector<int>& v, int target) {
    int left = 0, right = v.size() - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (v[mid] == target) return mid;
        else if (v[mid] < target) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}

// 二分答案
int binary_answer(int left, int right, function<bool(int)> check) {
    int ans = 0;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (check(mid)) {
            ans = mid;
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    return ans;
}
```
