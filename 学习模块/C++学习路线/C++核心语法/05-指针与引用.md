# 指针与引用

## 概述

指针是C++的核心特性，用于直接操作内存地址。虽然算法竞赛中引用更常用，但理解指针对深入学习C++和实现某些数据结构至关重要。

## 指针基础

### 指针声明与使用

```cpp
#include <iostream>
using namespace std;

int main() {
    int x = 10;
    int* p = &x;  // p存储x的地址
    
    cout << "x的值: " << x << endl;
    cout << "x的地址: " << &x << endl;
    cout << "p的值: " << p << endl;
    cout << "*p的值: " << *p << endl;  // 解引用
    
    *p = 20;  // 通过指针修改x
    cout << "修改后x: " << x << endl;
    
    return 0;
}
```

### 指针运算

```cpp
#include <iostream>
using namespace std;

int main() {
    int arr[] = {10, 20, 30, 40, 50};
    int* p = arr;  // 数组名即首元素指针
    
    // 指针算术
    cout << *(p + 2) << endl;  // 输出30
    cout << *(p + 4) << endl;  // 输出50
    
    // 指针递增
    p++;  // 指向arr[1]
    cout << *p << endl;  // 输出20
    
    // 指针差值
    int* start = arr;
    int* end = arr + 5;
    cout << "Distance: " << end - start << endl;  // 输出5
    
    return 0;
}
```

### 指针与数组

```cpp
#include <iostream>
using namespace std;

void printArray(int* arr, int size) {
    for (int i = 0; i < size; i++) {
        cout << arr[i] << " ";  // 等价于 *(arr + i)
    }
    cout << endl;
}

int main() {
    int arr[] = {1, 2, 3, 4, 5};
    
    // 数组名可当作指针使用
    printArray(arr, 5);
    
    // 指针也可用下标访问
    int* p = arr;
    cout << p[2] << endl;  // 等价于 arr[2]
    
    return 0;
}
```

## 引用

### 引用基础

```cpp
#include <iostream>
using namespace std;

int main() {
    int x = 10;
    int& ref = x;  // ref是x的别名
    
    cout << "x = " << x << endl;
    cout << "ref = " << ref << endl;
    
    ref = 20;  // 通过引用修改x
    cout << "修改后x = " << x << endl;
    
    // 引用必须初始化，且不能重新绑定
    int y = 30;
    // ref = y;  // 这是赋值，不是重新绑定
    
    return 0;
}
```

### 引用作为函数参数

```cpp
#include <iostream>
using namespace std;

// 交换函数
void swap(int& a, int& b) {
    int temp = a;
    a = b;
    b = temp;
}

// 返回引用
int& findMax(int& a, int& b) {
    return (a > b) ? a : b;
}

int main() {
    int x = 10, y = 20;
    swap(x, y);
    cout << "x = " << x << ", y = " << y << endl;
    
    int& maxRef = findMax(x, y);
    maxRef = 100;  // 通过引用修改
    cout << "x = " << x << ", y = " << y << endl;
    
    return 0;
}
```

### const引用

```cpp
#include <iostream>
#include <string>
using namespace std;

// 只读引用，不能修改原值
void print(const string& s) {
    cout << s << endl;
    // s = "new";  // 错误：const引用不能修改
}

// 避免拷贝大对象
void process(const vector<int>& vec) {
    for (int x : vec) {
        cout << x << " ";
    }
    cout << endl;
}

int main() {
    string s = "Hello";
    print(s);
    
    return 0;
}
```

## 指针与动态内存

### new和delete

```cpp
#include <iostream>
using namespace std;

int main() {
    // 动态分配单个变量
    int* p = new int(10);
    cout << *p << endl;
    delete p;
    
    // 动态分配数组
    int* arr = new int[5]{1, 2, 3, 4, 5};
    for (int i = 0; i < 5; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
    delete[] arr;
    
    // 二维动态数组
    int rows = 3, cols = 4;
    int** matrix = new int*[rows];
    for (int i = 0; i < rows; i++) {
        matrix[i] = new int[cols]{0};
    }
    
    // 使用...
    
    // 释放
    for (int i = 0; i < rows; i++) {
        delete[] matrix[i];
    }
    delete[] matrix;
    
    return 0;
}
```

### 智能指针（C++11）

```cpp
#include <iostream>
#include <memory>
using namespace std;

int main() {
    // unique_ptr：独占所有权
    unique_ptr<int> p1 = make_unique<int>(10);
    cout << *p1 << endl;
    
    // shared_ptr：共享所有权
    shared_ptr<int> p2 = make_shared<int>(20);
    shared_ptr<int> p3 = p2;  // 引用计数+1
    cout << *p2 << " " << *p3 << endl;
    
    // weak_ptr：弱引用，不增加引用计数
    weak_ptr<int> wp = p2;
    if (auto sp = wp.lock()) {
        cout << *sp << endl;
    }
    
    return 0;
}
```

## 指针常见陷阱

### 悬垂指针

```cpp
int* dangling() {
    int x = 10;
    return &x;  // 错误：返回局部变量地址
}

// 正确做法
int* safe() {
    int* p = new int(10);
    return p;  // 调用者负责delete
}
```

### 内存泄漏

```cpp
void leak() {
    int* p = new int(10);
    // 忘记delete p
    // 函数结束时p被销毁，但内存未释放
}
```

### 空指针检查

```cpp
void process(int* p) {
    if (p == nullptr) {
        cout << "Invalid pointer" << endl;
        return;
    }
    cout << *p << endl;
}
```

## 练习题

### 题目1：原地移除元素
**题目描述**：给定一个数组和一个值，原地移除所有等于该值的元素，返回新长度。

**参考代码**：
```cpp
#include <iostream>
#include <vector>
using namespace std;

int removeElement(vector<int>& nums, int val) {
    int slow = 0;
    for (int fast = 0; fast < nums.size(); fast++) {
        if (nums[fast] != val) {
            nums[slow] = nums[fast];
            slow++;
        }
    }
    return slow;
}

int main() {
    vector<int> nums = {3, 2, 2, 3};
    int val = 3;
    
    int newLen = removeElement(nums, val);
    for (int i = 0; i < newLen; i++) {
        cout << nums[i] << " ";
    }
    cout << endl;
    
    return 0;
}
```

### 题目2：合并有序数组
**题目描述**：将两个有序数组合并为一个有序数组。

**参考代码**：
```cpp
#include <iostream>
#include <vector>
using namespace std;

vector<int> mergeSorted(vector<int>& nums1, vector<int>& nums2) {
    vector<int> result;
    int i = 0, j = 0;
    
    while (i < nums1.size() && j < nums2.size()) {
        if (nums1[i] < nums2[j]) {
            result.push_back(nums1[i++]);
        } else {
            result.push_back(nums2[j++]);
        }
    }
    
    while (i < nums1.size()) result.push_back(nums1[i++]);
    while (j < nums2.size()) result.push_back(nums2[j++]);
    
    return result;
}

int main() {
    vector<int> nums1 = {1, 3, 5};
    vector<int> nums2 = {2, 4, 6};
    
    vector<int> merged = mergeSorted(nums1, nums2);
    for (int x : merged) cout << x << " ";
    cout << endl;
    
    return 0;
}
```

## 小结

1. 引用比指针更安全，优先使用引用
2. 函数参数只读时使用`const`引用
3. 动态内存用`new/delete`，优先使用智能指针
4. 注意避免悬垂指针和内存泄漏
