# 哈希表

## 概述

哈希表通过哈希函数将键映射到数组索引，实现O(1)的平均查找、插入和删除。是算法中最重要的数据结构之一。

## 哈希表基础

### 链地址法实现

```cpp
#include <iostream>
#include <vector>
#include <list>
using namespace std;

class HashTable {
private:
    vector<list<pair<int, int>>> table;
    int capacity;
    int size;
    
    int hash(int key) {
        return abs(key) % capacity;
    }
    
public:
    HashTable(int cap = 16) : capacity(cap), size(0) {
        table.resize(capacity);
    }
    
    void put(int key, int value) {
        int index = hash(key);
        
        for (auto& [k, v] : table[index]) {
            if (k == key) {
                v = value;
                return;
            }
        }
        
        table[index].push_back({key, value});
        size++;
    }
    
    int get(int key) {
        int index = hash(key);
        
        for (auto& [k, v] : table[index]) {
            if (k == key) return v;
        }
        
        return -1;  // 未找到
    }
    
    void remove(int key) {
        int index = hash(key);
        auto& chain = table[index];
        
        for (auto it = chain.begin(); it != chain.end(); it++) {
            if (it->first == key) {
                chain.erase(it);
                size--;
                return;
            }
        }
    }
    
    bool contains(int key) {
        return get(key) != -1;
    }
    
    int getSize() {
        return size;
    }
};

int main() {
    HashTable ht;
    ht.put(1, 10);
    ht.put(2, 20);
    ht.put(3, 30);
    
    cout << "Get 1: " << ht.get(1) << endl;
    cout << "Get 2: " << ht.get(2) << endl;
    
    ht.put(2, 25);
    cout << "Get 2 after update: " << ht.get(2) << endl;
    
    ht.remove(1);
    cout << "Get 1 after remove: " << ht.get(1) << endl;
    
    return 0;
}
```

### 开放寻址法实现

```cpp
#include <iostream>
#include <vector>
using namespace std;

class OpenAddressHashTable {
private:
    vector<int> keys;
    vector<int> values;
    vector<bool> occupied;
    int capacity;
    int size;
    
    int hash(int key) {
        return abs(key) % capacity;
    }
    
    int probe(int index, int i) {
        return (index + i) % capacity;
    }
    
public:
    OpenAddressHashTable(int cap = 16) : capacity(cap), size(0) {
        keys.resize(capacity, -1);
        values.resize(capacity, -1);
        occupied.resize(capacity, false);
    }
    
    void put(int key, int value) {
        if (size >= capacity * 0.7) {
            // 负载因子过高，需要扩容
            cout << "Hash table is too full!" << endl;
            return;
        }
        
        int index = hash(key);
        int i = 0;
        
        while (i < capacity) {
            int pos = probe(index, i);
            
            if (!occupied[pos] || keys[pos] == key) {
                keys[pos] = key;
                values[pos] = value;
                occupied[pos] = true;
                if (i == 0 || keys[pos] != key) size++;
                return;
            }
            
            i++;
        }
    }
    
    int get(int key) {
        int index = hash(key);
        int i = 0;
        
        while (i < capacity) {
            int pos = probe(index, i);
            
            if (!occupied[pos]) return -1;
            if (keys[pos] == key) return values[pos];
            
            i++;
        }
        
        return -1;
    }
    
    void remove(int key) {
        int index = hash(key);
        int i = 0;
        
        while (i < capacity) {
            int pos = probe(index, i);
            
            if (!occupied[pos]) return;
            
            if (keys[pos] == key) {
                occupied[pos] = false;
                keys[pos] = -1;
                values[pos] = -1;
                size--;
                return;
            }
            
            i++;
        }
    }
};

int main() {
    OpenAddressHashTable ht;
    ht.put(1, 100);
    ht.put(2, 200);
    ht.put(3, 300);
    
    cout << "Get 1: " << ht.get(1) << endl;
    cout << "Get 2: " << ht.get(2) << endl;
    
    return 0;
}
```

## STL哈希容器

### unordered_map

```cpp
#include <iostream>
#include <unordered_map>
#include <string>
using namespace std;

int main() {
    unordered_map<string, int> um;
    
    // 插入
    um["apple"] = 3;
    um["banana"] = 5;
    um.insert({"orange", 2});
    
    // 查找
    if (um.count("apple")) {
        cout << "apple: " << um["apple"] << endl;
    }
    
    // 遍历
    for (auto& [key, value] : um) {
        cout << key << ": " << value << endl;
    }
    
    // 删除
    um.erase("banana");
    
    cout << "Size: " << um.size() << endl;
    
    return 0;
}
```

### unordered_set

```cpp
#include <iostream>

using namespace std;

int main() {
    unordered_set<int> us;

    for (int x : us) cout << x << " ";

    return 0;
}
```

### 哈希表用于计数

```cpp
#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

// 字符频率统计
void countFrequency(string s) {
    unordered_map<char, int> freq;
    
    for (char c : s) {
        freq[c]++;
    }
    
    for (auto& [ch, count] : freq) {
        cout << ch << ": " << count << endl;
    }
}

int main() {
    countFrequency("hello world");
    return 0;
}
```

## 哈希表的经典应用

### 两数之和

```cpp
#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

vector<int> twoSum(vector<int>& nums, int target) {
    unordered_map<int, int> map;
    
    for (int i = 0; i < nums.size(); i++) {
        int complement = target - nums[i];
        if (map.count(complement)) {
            return {map[complement], i};
        }
        map[nums[i]] = i;
    }
    
    return {};
}

int main() {
    vector<int> nums = {2, 7, 11, 15};
    int target = 9;
    
    auto result = twoSum(nums, target);
    cout << result[0] << ", " << result[1] << endl;
    
    return 0;
}
```

### 字母异位词分组

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <unordered_map>
#include <algorithm>
using namespace std;

vector<vector<string>> groupAnagrams(vector<string>& strs) {
    unordered_map<string, vector<string>> map;
    
    for (string s : strs) {
        string key = s;
        sort(key.begin(), key.end());
        map[key].push_back(s);
    }
    
    vector<vector<string>> result;
    for (auto& [key, group] : map) {
        result.push_back(group);
    }
    
    return result;
}

int main() {
    vector<string> strs = {"eat", "tea", "tan", "ate", "nat", "bat"};
    
    auto result = groupAnagrams(strs);
    for (auto& group : result) {
        for (string s : group) cout << s << " ";
        cout << endl;
    }
    
    return 0;
}
```

### 最长连续序列

```cpp
#include <iostream>
#include <vector>
#include <unordered_set>
using namespace std;

int longestConsecutive(vector<int>& nums) {
    unordered_set<int> numSet(nums.begin(), nums.end());
    int longest = 0;
    
    for (int num : numSet) {
        // 只从序列起点开始计数
        if (numSet.count(num - 1)) continue;
        
        int currentNum = num;
        int currentStreak = 1;
        
        while (numSet.count(currentNum + 1)) {
            currentNum++;
            currentStreak++;
        }
        
        longest = max(longest, currentStreak);
    }
    
    return longest;
}

int main() {
    vector<int> nums = {100, 4, 200, 1, 3, 2};
    cout << longestConsecutive(nums) << endl;  // 4
    
    return 0;
}
```

## 哈希冲突处理

### 常见哈希函数

```cpp
// 整数哈希
int hashInt(int key, int capacity) {
    return abs(key) % capacity;
}

// 字符串哈希
int hashString(string s, int capacity) {
    long long hash = 0;
    long long base = 31;
    long long mod = 1e9 + 7;
    
    for (char c : s) {
        hash = (hash * base + (c - 'a' + 1)) % mod;
    }
    
    return hash % capacity;
}
```

## 练习题

### 题目1：有效的字母异位词
**题目描述**：判断两个字符串是否为字母异位词。

**参考代码**：
```cpp
#include <iostream>
#include <string>
#include <vector>
using namespace std;

bool isAnagram(string s, string t) {
    if (s.length() != t.length()) return false;
    
    vector<int> count(26, 0);
    
    for (char c : s) count[c - 'a']++;
    for (char c : t) count[c - 'a']--;
    
    for (int x : count) {
        if (x != 0) return false;
    }
    
    return true;
}

int main() {
    cout << isAnagram("anagram", "nagaram") << endl;  // 1
    cout << isAnagram("rat", "car") << endl;          // 0
    
    return 0;
}
```

### 题目2：LRU缓存
**题目描述**：设计并实现一个LRU（最近最少使用）缓存机制。

**参考代码**：
```cpp
#include <iostream>
#include <unordered_map>
#include <list>
using namespace std;

class LRUCache {
private:
    int capacity;
    list<pair<int, int>> cache;
    unordered_map<int, list<pair<int, int>>::iterator> map;
    
public:
    LRUCache(int cap) : capacity(cap) {}
    
    int get(int key) {
        if (map.find(key) == map.end()) return -1;
        
        // 移到头部
        cache.splice(cache.begin(), cache, map[key]);
        return map[key]->second;
    }
    
    void put(int key, int value) {
        if (map.find(key) != map.end()) {
            // 更新值并移到头部
            map[key]->second = value;
            cache.splice(cache.begin(), cache, map[key]);
            return;
        }
        
        if (cache.size() >= capacity) {
            // 删除尾部
            map.erase(cache.back().first);
            cache.pop_back();
        }
        
        // 插入头部
        cache.push_front({key, value});
        map[key] = cache.begin();
    }
};

int main() {
    LRUCache lru(2);
    lru.put(1, 1);
    lru.put(2, 2);
    cout << lru.get(1) << endl;  // 1
    
    lru.put(3, 3);  // 淘汰key=2
    cout << lru.get(2) << endl;  // -1
    
    return 0;
}
```

## 小结

1. 哈希表平均O(1)查找、插入、删除
2. 选择合适的哈希函数和处理冲突策略很重要
3. unordered_map/set是最常用的哈希容器
4. 哈希表适合快速查找、计数、分组等场景
