## 1. 双指针

**核心思想**：用两个指针遍历数组/链表，将 O(n²) 优化为 O(n)

### 三种变体

| 变体 | 指针方向 | 典型应用 |
| ------ | ---------- | ---------- |
| 快慢指针 | 同向，速度不同 | 链表环检测、找中间节点 |
| 对撞指针 | 相向而行 | 两数之和、回文判断 |
| 同向指针 | 同向，同步移动 | 去重、子数组问题 |

```cpp
// 快慢指针：判断链表是否有环
bool hasCycle(ListNode* head) {
    ListNode *slow = head, *fast = head;
    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) return true;
    }
    return false;
}

// 对撞指针：两数之和（已排序）
vector<int> twoSum(vector<int>& numbers, int target) {
    int left = 0, right = numbers.size() - 1;
    while (left < right) {
        int sum = numbers[left] + numbers[right];
        if (sum == target) return {left + 1, right + 1};
        else if (sum < target) left++;
        else right--;
    }
    return {};
}

// 同向指针：移除重复项
int removeDuplicates(vector<int>& nums) {
    int slow = 0;
    for (int fast = 1; fast < nums.size(); fast++) {
        if (nums[fast] != nums[slow]) {
            slow++;
            nums[slow] = nums[fast];
        }
    }
    return slow + 1;
}
```

**练习**：LeetCode 26、11、15、42

## 2. 前缀和与差分

**前缀和**：区间和查询 O(1)
**差分数组**：区间修改 O(1)

```cpp
// 一维前缀和
vector<int> prefix(n + 1, 0);
for (int i = 0; i < n; i++)
    prefix[i + 1] = prefix[i] + nums[i];
int rangeSum = prefix[r + 1] - prefix[l];  // [l, r] 区间和

// 差分数组：区间 [l, r] 加 v
vector<int> diff(n + 1, 0);
diff[l] += v;
diff[r + 1] -= v;
```

| 数据结构 | 区间查询 | 区间修改 |
|----------|----------|----------|
| 前缀和 | O(1) | O(n) |
| 差分数组 | O(n) | O(1) |

## 3. 滑动窗口

**核心思想**：维护窗口 [left, right]，动态调整边界

```cpp
// 通用模板
int slidingWindow(vector<int>& nums, int target) {
    int left = 0, windowSum = 0, result = 0;

    for (int right = 0; right < nums.size(); right++) {
        windowSum += nums[right];

        while (windowSum >= target) {
            result = max(result, right - left + 1);
            windowSum -= nums[left];
            left++;
        }
    }
    return result;
}

// 最长无重复子串
int lengthOfLongestSubstring(string s) {
    unordered_map<char, int> map;
    int maxLen = 0, left = 0;

    for (int right = 0; right < s.size(); right++) {
        if (map.count(s[right]))
            left = max(left, map[s[right]] + 1);
        map[s[right]] = right;
        maxLen = max(maxLen, right - left + 1);
    }
    return maxLen;
}
```

## 4. 单调栈

**核心思想**：维护单调递增/递减栈，查找下一个更大/更小元素

```cpp
// 下一个更大元素（单调递减栈）
vector<int> nextGreaterElement(vector<int>& nums) {
    int n = nums.size();
    vector<int> result(n, -1);
    stack<int> stk;

    for (int i = 0; i < n; i++) {
        while (!stk.empty() && nums[stk.top()] < nums[i]) {
            result[stk.top()] = nums[i];
            stk.pop();
        }
        stk.push(i);
    }
    return result;
}
```

| 问题类型 | 栈类型 |
| ---------- | -------- |
| 下一个更大元素 | 单调递减栈 |
| 下一个更小元素 | 单调递增栈 |
| 柱状图最大矩形 | 单调递增栈 |

## 5. 并查集

**核心思想**：处理动态连通性问题

```cpp
class UnionFind {
    vector<int> parent, rank;
public:
    UnionFind(int n) : parent(n), rank(n, 0) {
        for (int i = 0; i < n; i++) parent[i] = i;
    }

    int find(int x) {
        if (parent[x] != x)
            parent[x] = find(parent[x]);  // 路径压缩
        return parent[x];
    }

    void unite(int x, int y) {
        int px = find(x), py = find(y);
        if (px == py) return;
        if (rank[px] < rank[py]) swap(px, py);
        parent[py] = px;
        if (rank[px] == rank[py]) rank[px]++;
    }

    bool connected(int x, int y) {
        return find(x) == find(y);
    }
};
```

if (n & 1) { }              // 判断奇偶

## 7. 离散化

**核心思想**：将大范围值映射到小范围

```cpp
vector<int> discretize(vector<int>& nums) {
    vector<int> sorted = nums;
    sort(sorted.begin(), sorted.end());
    sorted.erase(unique(sorted.begin(), sorted.end()), sorted.end());

    vector<int> result;
    for (int x : nums)
        result.push_back(lower_bound(sorted.begin(), sorted.end(), x) - sorted.begin());
    return result;
}
// 例: [100, 50, 200, 50] -> [1, 0, 2, 0]
```
