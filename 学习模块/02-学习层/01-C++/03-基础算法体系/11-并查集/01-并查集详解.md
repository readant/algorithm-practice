# 并查集

## 核心思想

处理动态连通性问题：快速判断两个元素是否在同一集合，以及合并两个集合。

---

## 基础实现

```cpp
class UnionFind {
private:
    vector<int> parent;

public:
    UnionFind(int n) : parent(n) {
        for (int i = 0; i < n; i++) parent[i] = i;
    }

    // 查找根节点
    int find(int x) {
        if (parent[x] != x) {
            parent[x] = find(parent[x]);  // 路径压缩
        }
        return parent[x];
    }

    // 合并两个集合
    void unite(int x, int y) {
        int px = find(x), py = find(y);
        if (px == py) return;
        parent[px] = py;
    }

    // 判断是否连通
    bool connected(int x, int y) {
        return find(x) == find(y);
    }
};
```

## 优化版本（按秩合并）

```cpp
class UnionFind {
private:
    vector<int> parent, rank;

public:
    UnionFind(int n) : parent(n), rank(n, 0) {
        for (int i = 0; i < n; i++) parent[i] = i;
    }

    int find(int x) {
        if (parent[x] != x) {
            parent[x] = find(parent[x]);
        }
        return parent[x];
    }

    void unite(int x, int y) {
        int px = find(x), py = find(y);
        if (px == py) return;

        // 按秩合并：矮树挂到高树下
        if (rank[px] < rank[py]) swap(px, py);
        parent[py] = px;
        if (rank[px] == rank[py]) rank[px]++;
    }

    bool connected(int x, int y) {
        return find(x) == find(y);
    }
};
```

## 经典应用

### 岛屿数量

```cpp
int numIslands(vector<vector<char>>& grid) {
    int m = grid.size(), n = grid[0].size();
    UnionFind uf(m * n);
    int count = 0;

    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            if (grid[i][j] == '0') continue;
            count++;

            // 合并相邻的陆地
            int dir[4][2] = {{0,1},{0,-1},{1,0},{-1,0}};
            for (auto& d : dir) {
                int ni = i + d[0], nj = j + d[1];
                if (ni >= 0 && ni < m && nj >= 0 && nj < n
                    && grid[ni][nj] == '1'
                    && !uf.connected(i * n + j, ni * n + nj)) {
                    uf.unite(i * n + j, ni * n + nj);
                    count--;
                }
            }
        }
    }

    return count;
}
```

### 冗余连接

```cpp
vector<int> findRedundantDirectedConnection(vector<vector<int>>& edges) {
    int n = edges.size();
    UnionFind uf(n + 1);

    for (auto& edge : edges) {
        if (uf.connected(edge[0], edge[1])) {
            return edge;
        }
        uf.unite(edge[0], edge[1]);
    }

    return {};
}
```

## 复杂度

| 操作 | 时间复杂度 |
|------|-----------|
| find | O(α(n)) ≈ O(1) |
| unite | O(α(n)) ≈ O(1) |
| connected | O(α(n)) ≈ O(1) |

其中 α(n) 是反阿克曼函数，增长极慢。
