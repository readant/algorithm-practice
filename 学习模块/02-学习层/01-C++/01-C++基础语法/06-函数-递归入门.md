# 函数与递归

## 概述

函数是代码复用的基本单元。递归是算法设计的核心思想之一，许多经典算法（如归并排序、快速排序、DFS）都基于递归实现。

## 函数基础

### 函数定义与调用

```cpp
#include <iostream>
using namespace std;

// 函数声明（可选，定义在main之前可省略）
int add(int a, int b);
void printArray(int arr[], int size);

int main() {
    int result = add(3, 5);
    cout << "3 + 5 = " << result << endl;
    
    int arr[] = {1, 2, 3, 4, 5};
    printArray(arr, 5);
    
    return 0;
}

// 函数定义
int add(int a, int b) {
    return a + b;
}

void printArray(int arr[], int size) {
    for (int i = 0; i < size; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
}
```

### 参数传递

```cpp
#include <iostream>
using namespace std;

// 值传递（不修改原变量）
void swapByValue(int a, int b) {
    int temp = a;
    a = b;
    b = temp;
}

// 引用传递（可修改原变量）
void swapByReference(int &a, int &b) {
    int temp = a;
    a = b;
    b = temp;
}

// 指针传递
void swapByPointer(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main() {
    int x = 10, y = 20;
    
    swapByValue(x, y);
    cout << "After swapByValue: x=" << x << ", y=" << y << endl;
    
    swapByReference(x, y);
    cout << "After swapByReference: x=" << x << ", y=" << y << endl;
    
    swapByPointer(&x, &y);
    cout << "After swapByPointer: x=" << x << ", y=" << y << endl;
    
    return 0;
}
```

### 数组作为参数

```cpp
#include <iostream>
using namespace std;

// 数组作为参数时会退化为指针
void modifyArray(int arr[], int size) {
    for (int i = 0; i < size; i++) {
        arr[i] *= 2;  // 会修改原数组
    }
}

// 二维数组
void print2DArray(int arr[][3], int rows) {
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < 3; j++) {
            cout << arr[i][j] << " ";
        }
        cout << endl;
    }
}

int main() {
    int arr[] = {1, 2, 3, 4, 5};
    modifyArray(arr, 5);
    
    for (int x : arr) cout << x << " ";
    cout << endl;
    
    return 0;
}
```

## 递归

### 递归基础

递归函数必须包含：
1. **基准情况（Base Case）**：停止递归的条件
2. **递归情况（Recursive Case）**：调用自身缩小问题规模

```cpp
#include <iostream>
using namespace std;

// 阶乘
long long factorial(int n) {
    if (n <= 1) return 1;  // 基准情况
    return n * factorial(n - 1);  // 递归情况
}

// 斐波那契数列
int fibonacci(int n) {
    if (n <= 1) return n;
    return fibonacci(n - 1) + fibonacci(n - 2);
}

// 汉诺塔
void hanoi(int n, char from, char to, char aux) {
    if (n == 1) {
        cout << "Move disk 1 from " << from << " to " << to << endl;
        return;
    }
    hanoi(n - 1, from, aux, to);
    cout << "Move disk " << n << " from " << to << " to " << to << endl;
    hanoi(n - 1, aux, to, from);
}

int main() {
    cout << "5! = " << factorial(5) << endl;
    cout << "fib(10) = " << fibonacci(10) << endl;
    
    return 0;
}
```

### 递归优化：记忆化

```cpp
#include <iostream>
#include <cstring>
using namespace std;

const int MAXN = 100;
long long memo[MAXN];

// 记忆化斐波那契 O(n)
long long fibMemo(int n) {
    if (n <= 1) return n;
    if (memo[n] != -1) return memo[n];
    memo[n] = fibMemo(n - 1) + fibMemo(n - 2);
    return memo[n];
}

int main() {
    memset(memo, -1, sizeof(memo));
    cout << "fib(40) = " << fibMemo(40) << endl;
    return 0;
}
```

### 递归转迭代

```cpp
#include <iostream>
using namespace std;

// 递归版本
long long fibRecursive(int n) {
    if (n <= 1) return n;
    return fibRecursive(n - 1) + fibRecursive(n - 2);
}

// 迭代版本 O(n)时间 O(1)空间
long long fibIterative(int n) {
    if (n <= 1) return n;
    long long a = 0, b = 1;
    for (int i = 2; i <= n; i++) {
        long long c = a + b;
        a = b;
        b = c;
    }
    return b;
}

int main() {
    cout << "Recursive: " << fibRecursive(10) << endl;
    cout << "Iterative: " << fibIterative(10) << endl;
    return 0;
}
```

## 内联函数

```cpp
#include <iostream>
using namespace std;

// 内联函数：编译器会将函数体展开到调用处，减少函数调用开销
inline int square(int x) {
    return x * x;
}

inline int max3(int a, int b, int c) {
    return max(a, max(b, c));
}

int main() {
    cout << square(5) << endl;
    cout << max3(10, 20, 30) << endl;
    return 0;
}
```

## 练习题

### 题目1：递归求幂
**题目描述**：实现 pow(x, n) 函数，计算x的n次幂。

**参考代码**：
```cpp
#include <iostream>
using namespace std;

// 朴素递归 O(n)
double myPow(double x, long long n) {
    if (n == 0) return 1;
    if (n < 0) {
        x = 1 / x;
        n = -n;
    }
    return x * myPow(x, n - 1);
}

// 快速幂 O(log n)
double fastPow(double x, long long n) {
    if (n == 0) return 1;
    if (n < 0) {
        x = 1 / x;
        n = -n;
    }
    
    double result = 1.0;
    while (n > 0) {
        if (n % 2 == 1) {
            result *= x;
        }
        x *= x;
        n /= 2;
    }
    return result;
}

int main() {
    cout << myPow(2, 10) << endl;    // 1024
    cout << fastPow(2, 10) << endl;  // 1024
    return 0;
}
```

### 题目2：全排列
**题目描述**：给定一个不含重复数字的数组，返回其所有可能的全排列。

**参考代码**：
```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

vector<vector<int>> result;

void permute(vector<int>& nums, int start) {
    if (start == nums.size()) {
        result.push_back(nums);
        return;
    }
    
    for (int i = start; i < nums.size(); i++) {
        swap(nums[start], nums[i]);
        permute(nums, start + 1);
        swap(nums[start], nums[i]);  // 回溯
    }
}

int main() {
    vector<int> nums = {1, 2, 3};
    permute(nums, 0);
    
    for (auto& perm : result) {
        for (int x : perm) cout << x << " ";
        cout << endl;
    }
    
    return 0;
}
```

## 小结

1. 引用传递可修改原变量，值传递会复制
2. 递归必须有基准情况，否则会栈溢出
3. 递归可转迭代，通常迭代更高效
4. 记忆化可将指数级递归优化为多项式级
