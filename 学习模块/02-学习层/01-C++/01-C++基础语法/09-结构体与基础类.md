# 结构体与类

## 概述

结构体和类用于封装相关数据和操作。在算法实现中，常用于定义图的边、节点等复杂数据结构。

## 结构体

### 基本用法

```cpp
#include <iostream>
#include <string>
using namespace std;

struct Student {
    string name;
    int age;
    double score;
    
    // 成员函数
    void print() {
        cout << name << ", " << age << ", " << score << endl;
    }
};

int main() {
    // 创建结构体变量
    Student s1 = {"Alice", 20, 95.5};
    Student s2;
    s2.name = "Bob";
    s2.age = 21;
    s2.score = 88.0;
    
    s1.print();
    s2.print();
    
    return 0;
}
```

### 结构体数组

```cpp
#include <iostream>
#include <string>
using namespace std;

struct Point {
    int x, y;
};

int main() {
    // 结构体数组
    Point points[3] = {{1, 2}, {3, 4}, {5, 6}};
    
    for (int i = 0; i < 3; i++) {
        cout << "(" << points[i].x << ", " << points[i].y << ") ";
    }
    cout << endl;
    
    return 0;
}
```

### 结构体指针

```cpp
#include <iostream>
using namespace std;

struct Node {
    int data;
    Node* next;
};

int main() {
    Node* head = new Node{1, nullptr};
    head->next = new Node{2, nullptr};
    head->next->next = new Node{3, nullptr};
    
    Node* curr = head;
    while (curr != nullptr) {
        cout << curr->data << " -> ";
        curr = curr->next;
    }
    cout << "NULL" << endl;
    
    return 0;
}
```

## 类

### 基本类定义

```cpp
#include <iostream>
#include <string>
using namespace std;

class Person {
private:
    string name;
    int age;
    
public:
    // 构造函数
    Person(const string& n, int a) : name(n), age(a) {}
    
    // 成员函数
    void introduce() const {
        cout << "I'm " << name << ", " << age << " years old." << endl;
    }
    
    // getter/setter
    string getName() const { return name; }
    int getAge() const { return age; }
    void setAge(int a) { age = a; }
};

int main() {
    Person p("Charlie", 25);
    p.introduce();
    p.setAge(26);
    p.introduce();
    
    return 0;
}
```

### 运算符重载

```cpp
#include <iostream>
using namespace std;

class Vector2D {
private:
    double x, y;
    
public:
    Vector2D(double x = 0, double y = 0) : x(x), y(y) {}
    
    // 加法
    Vector2D operator+(const Vector2D& other) const {
        return Vector2D(x + other.x, y + other.y);
    }
    
    // 比较
    bool operator==(const Vector2D& other) const {
        return x == other.x && y == other.y;
    }
    
    // 输出
    friend ostream& operator<<(ostream& os, const Vector2D& v) {
        os << "(" << v.x << ", " << v.y << ")";
        return os;
    }
};

int main() {
    Vector2D v1(1, 2);
    Vector2D v2(3, 4);
    Vector2D v3 = v1 + v2;
    
    cout << v1 << " + " << v2 << " = " << v3 << endl;
    
    return 0;
}
```

### 自定义排序

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct Student {
    string name;
    int score;
};

int main() {
    vector<Student> students = {
        {"Alice", 90},
        {"Bob", 85},
        {"Charlie", 95}
    };
    
    // 按分数降序排序
    sort(students.begin(), students.end(), 
         [](const Student& a, const Student& b) {
             return a.score > b.score;
         });
    
    for (const auto& s : students) {
        cout << s.name << ": " << s.score << endl;
    }
    
    return 0;
}
```

### 并查集

```cpp
#include <iostream>
#include <vector>
using namespace std;

class UnionFind {
private:
    vector<int> parent;
    vector<int> rank;
    
public:
    UnionFind(int n) : parent(n), rank(n, 0) {
        for (int i = 0; i < n; i++) {
            parent[i] = i;
        }
    }
    
    int find(int x) {
        if (parent[x] != x) {
            parent[x] = find(parent[x]);  // 路径压缩
        }
        return parent[x];
    }
    
    void unite(int x, int y) {
        int px = find(x), py = find(y);
        if (px == py) return;
        
        if (rank[px] < rank[py]) swap(px, py);
        parent[py] = px;
        if (rank[px] == rank[py]) rank[px]++;
    }
    
    bool connected(int x, int y) {
        return find(x) == find(y);
    }
};

int main() {
    UnionFind uf(6);
    uf.unite(0, 1);
    uf.unite(1, 2);
    uf.unite(3, 4);
    
    cout << "0 and 2 connected: " << uf.connected(0, 2) << endl;
    cout << "0 and 3 connected: " << uf.connected(0, 3) << endl;
    
    return 0;
}
```

## 练习题

### 题目1：定义链表
**题目描述**：实现一个简单的单链表，支持插入、删除、打印操作。

**参考代码**：
```cpp
#include <iostream>
using namespace std;

struct ListNode {
    int val;
    ListNode* next;
    ListNode(int x) : val(x), next(nullptr) {}
};

class LinkedList {
private:
    ListNode* head;
    
public:
    LinkedList() : head(nullptr) {}
    
    void insert(int val) {
        ListNode* node = new ListNode(val);
        node->next = head;
        head = node;
    }
    
    void remove(int val) {
        if (!head) return;
        
        if (head->val == val) {
            ListNode* temp = head;
            head = head->next;
            delete temp;
            return;
        }
        
        ListNode* curr = head;
        while (curr->next && curr->next->val != val) {
            curr = curr->next;
        }
        
        if (curr->next) {
            ListNode* temp = curr->next;
            curr->next = temp->next;
            delete temp;
        }
    }
    
    void print() {
        ListNode* curr = head;
        while (curr) {
            cout << curr->val << " -> ";
            curr = curr->next;
        }
        cout << "NULL" << endl;
    }
};

int main() {
    LinkedList list;
    list.insert(3);
    list.insert(2);
    list.insert(1);
    list.print();
    
    list.remove(2);
    list.print();
    
    return 0;
}
```

### 题目2：定义图
**题目描述**：使用邻接表实现图的创建和遍历。

**参考代码**：
```cpp
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

class Graph {
private:
    int V;
    vector<vector<int>> adj;
    
public:
    Graph(int V) : V(V), adj(V) {}
    
    void addEdge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);  // 无向图
    }
    
    void bfs(int start) {
        vector<bool> visited(V, false);
        queue<int> q;
        
        visited[start] = true;
        q.push(start);
        
        while (!q.empty()) {
            int v = q.front();
            q.pop();
            cout << v << " ";
            
            for (int u : adj[v]) {
                if (!visited[u]) {
                    visited[u] = true;
                    q.push(u);
                }
            }
        }
        cout << endl;
    }
    
    void dfs(int v, vector<bool>& visited) {
        visited[v] = true;
        cout << v << " ";
        
        for (int u : adj[v]) {
            if (!visited[u]) {
                dfs(u, visited);
            }
        }
    }
};

int main() {
    Graph g(5);
    g.addEdge(0, 1);
    g.addEdge(0, 4);
    g.addEdge(1, 2);
    g.addEdge(1, 3);
    g.addEdge(2, 3);
    
    cout << "BFS from 0: ";
    g.bfs(0);
    
    cout << "DFS from 0: ";
    vector<bool> visited(5, false);
    g.dfs(0, visited);
    cout << endl;
    
    return 0;
}
```

## 小结

1. 结构体适合简单的数据聚合
2. 类适合需要封装和抽象的复杂数据
3. 运算符重载可让自定义类型像内置类型一样使用
4. 算法中常用结构体表示图的边、节点等
