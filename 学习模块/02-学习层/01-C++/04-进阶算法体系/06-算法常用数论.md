## 2. GCD 与 LCM

```cpp
// 欧几里得算法
int gcd(int a, int b) {
    return b == 0 ? a : gcd(b, a % b);
}

// 最小公倍数
int lcm(int a, int b) {
    return a / gcd(a, b) * b;
}

// C++17 内置
#include <numeric>
int g = gcd(a, b);
int l = lcm(a, b);
```

## 3. 质数相关

### 判断质数

```cpp
bool isPrime(int n) {
    if (n < 2) return false;
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) return false;
    }
    return true;
}
```

### 埃氏筛（找范围内所有质数）

```cpp
vector<int> sieve(int n) {
    vector<bool> isPrime(n + 1, true);
    vector<int> primes;
    
    for (int i = 2; i <= n; i++) {
        if (isPrime[i]) {
            primes.push_back(i);
            for (int j = i * i; j <= n; j += i) {
                isPrime[j] = false;
            }
        }
    }
    
    return primes;
}
```

### 质因数分解

```cpp
vector<pair<int, int>> factorize(int n) {
    vector<pair<int, int>> factors;
    
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            int count = 0;
            while (n % i == 0) {
                n /= i;
                count++;
            }
            factors.push_back({i, count});
        }
    }
    
    if (n > 1) factors.push_back({n, 1});
    
    return factors;
}
```

## 4. 快速幂

```cpp
long long power(long long base, long long exp, long long mod) {
    long long result = 1;
    base %= mod;
    
    while (exp > 0) {
        if (exp & 1) {
            result = result * base % mod;
        }
        base = base * base % mod;
        exp >>= 1;
    }
    
    return result;
}
```

## 5. 模逆元

### 费马小定理（mod为质数）

```cpp
long long modInverse(long long a, long long mod) {
    return power(a, mod - 2, mod);
}
```

### 扩展欧几里得

```cpp
int extgcd(int a, int b, int& x, int& y) {
    if (b == 0) {
        x = 1; y = 0;
        return a;
    }
    
    int x1, y1;
    int g = extgcd(b, a % b, x1, y1);
    
    x = y1;
    y = x1 - (a / b) * y1;
    
    return g;
}
```

## 6. 组合数学

### 预处理阶乘

```cpp
const int MOD = 1e9 + 7;
const int MAXN = 1e6 + 5;

long long fact[MAXN], invFact[MAXN];

void precompute() {
    fact[0] = 1;
    for (int i = 1; i < MAXN; i++) {
        fact[i] = fact[i - 1] * i % MOD;
    }
    invFact[MAXN - 1] = power(fact[MAXN - 1], MOD - 2, MOD);
    for (int i = MAXN - 2; i >= 0; i--) {
        invFact[i] = invFact[i + 1] * (i + 1) % MOD;
    }
}

// 组合数 C(n, k)
long long comb(int n, int k) {
    if (k < 0 || k > n) return 0;
    return fact[n] * invFact[k] % MOD * invFact[n - k] % MOD;
}
```

### Lucas定理（大组合数取模）

```cpp
long long lucas(long long n, long long k, long long p) {
    if (k == 0) return 1;
    return comb(n % p, k % p, p) * lucas(n / p, k / p, p) % p;
}
```

## 7. 博弈论基础

### Nim游戏

```cpp
// 判断先手是否必胜
bool nimGame(vector<int>& piles) {
    int xorSum = 0;
    for (int pile : piles) {
        xorSum ^= pile;
    }
    return xorSum != 0;
}
```
