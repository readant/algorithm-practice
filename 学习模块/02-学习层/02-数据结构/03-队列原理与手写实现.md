# 队列

## 概述

队列是一种先进先出（FIFO）的线性数据结构，只允许在队尾插入、队头删除。广泛应用于广度优先搜索、任务调度、缓冲区等场景。

## 队列的基本实现

### 数组实现（循环队列）

```cpp
#include <iostream>
using namespace std;

class CircularQueue {
private:
    int data[1000];
    int front, rear, capacity;
    
public:
    CircularQueue(int cap) : front(0), rear(-1), capacity(cap) {}
    
    void enqueue(int val) {
        if (isFull()) {
            cout << "Queue is full!" << endl;
            return;
        }
        rear = (rear + 1) % capacity;
        data[rear] = val;
    }
    
    int dequeue() {
        if (isEmpty()) {
            cout << "Queue is empty!" << endl;
            return -1;
        }
        int val = data[front];
        front = (front + 1) % capacity;
        return val;
    }
    
    int peek() {
        if (isEmpty()) return -1;
        return data[front];
    }
    
    bool isEmpty() {
        return front > rear;
    }
    
    bool isFull() {
        return (rear + 1) % capacity == front;
    }
    
    int size() {
        if (isEmpty()) return 0;
        return (rear - front + capacity) % capacity + 1;
    }
};

int main() {
    CircularQueue q(5);
    q.enqueue(1);
    q.enqueue(2);
    q.enqueue(3);
    
    cout << "Front: " << q.peek() << endl;
    cout << "Size: " << q.size() << endl;
    
    while (!q.isEmpty()) {
        cout << q.dequeue() << " ";
    }
    cout << endl;
    
    return 0;
}
```

### 链表实现

```cpp
#include <iostream>
using namespace std;

struct Node {
    int val;
    Node* next;
    Node(int x) : val(x), next(nullptr) {}
};

class LinkedQueue {
private:
    Node* frontNode;
    Node* rearNode;
    int queueSize;
    
public:
    LinkedQueue() : frontNode(nullptr), rearNode(nullptr), queueSize(0) {}
    
    ~LinkedQueue() {
        while (!isEmpty()) dequeue();
    }
    
    void enqueue(int val) {
        Node* node = new Node(val);
        if (isEmpty()) {
            frontNode = rearNode = node;
        } else {
            rearNode->next = node;
            rearNode = node;
        }
        queueSize++;
    }
    
    int dequeue() {
        if (isEmpty()) return -1;
        
        int val = frontNode->val;
        Node* temp = frontNode;
        frontNode = frontNode->next;
        if (!frontNode) rearNode = nullptr;
        delete temp;
        queueSize--;
        return val;
    }
    
    int peek() {
        return frontNode ? frontNode->val : -1;
    }
    
    bool isEmpty() {
        return frontNode == nullptr;
    }
    
    int size() {
        return queueSize;
    }
};

int main() {
    LinkedQueue q;
    q.enqueue(10);
    q.enqueue(20);
    q.enqueue(30);
    
    cout << "Front: " << q.peek() << endl;
    
    while (!q.isEmpty()) {
        cout << q.dequeue() << " ";
    }
    cout << endl;
    
    return 0;
}
```

## STL队列

#include <iostream>

### 双端队列（deque）

```cpp
#include <iostream>
#include <deque>
using namespace std;

int main() {
    deque<int> dq;
    
    dq.push_back(1);
    dq.push_back(2);
    dq.push_front(0);
    
    cout << "Front: " << dq.front() << endl;
    cout << "Back: " << dq.back() << endl;
    
    dq.pop_front();
    dq.pop_back();
    
    for (int x : dq) cout << x << " ";
    cout << endl;
    
    return 0;
}
```

### 优先队列（堆）

```cpp
#include <iostream>
#include <queue>
using namespace std;

int main() {
    // 最大堆
    priority_queue<int> maxPq;
    maxPq.push(3);
    maxPq.push(1);
    maxPq.push(4);
    
    cout << "Max heap: ";
    while (!maxPq.empty()) {
        cout << maxPq.top() << " ";
        maxPq.pop();
    }
    cout << endl;
    
    // 最小堆
    priority_queue<int, vector<int>, greater<int>> minPq;
    minPq.push(3);
    minPq.push(1);
    minPq.push(4);
    
    cout << "Min heap: ";
    while (!minPq.empty()) {
        cout << minPq.top() << " ";
        minPq.pop();
    }
    cout << endl;
    
    return 0;
}
```

## 队列的经典应用

### 广度优先搜索（BFS）

```cpp
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

void bfs(vector<vector<int>>& graph, int start) {
    int n = graph.size();
    vector<bool> visited(n, false);
    queue<int> q;
    
    visited[start] = true;
    q.push(start);
    
    while (!q.empty()) {
        int v = q.front();
        q.pop();
        cout << v << " ";
        
        for (int u : graph[v]) {
            if (!visited[u]) {
                visited[u] = true;
                q.push(u);
            }
        }
    }
    cout << endl;
}

int main() {
    vector<vector<int>> graph = {
        {1, 2},    // 0的邻居
        {0, 3, 4}, // 1的邻居
        {0, 4},    // 2的邻居
        {1},       // 3的邻居
        {1, 2}     // 4的邻居
    };
    
    bfs(graph, 0);
    
    return 0;
}
```

### 滑动窗口最大值

```cpp
#include <iostream>
#include <vector>
#include <deque>
using namespace std;

vector<int> maxSlidingWindow(vector<int>& nums, int k) {
    vector<int> result;
    deque<int> dq;  // 存储索引，维护单调递减
    
    for (int i = 0; i < nums.size(); i++) {
        // 移除超出窗口的元素
        while (!dq.empty() && dq.front() < i - k + 1) {
            dq.pop_front();
        }
        
        // 保持单调递减
        while (!dq.empty() && nums[i] > nums[dq.back()]) {
            dq.pop_back();
        }
        
        dq.push_back(i);
        
        // 窗口形成后记录结果
        if (i >= k - 1) {
            result.push_back(nums[dq.front()]);
        }
    }
    
    return result;
}

int main() {
    vector<int> nums = {1, 3, -1, -3, 5, 3, 6, 7};
    int k = 3;
    
    vector<int> result = maxSlidingWindow(nums, k);
    for (int x : result) cout << x << " ";
    cout << endl;
    
    return 0;
}
```

### 任务调度

```cpp
#include <iostream>
#include <queue>
#include <vector>
using namespace std;

int leastInterval(vector<char>& tasks, int n) {
    vector<int> count(26, 0);
    for (char task : tasks) {
        count[task - 'A']++;
    }
    
    priority_queue<int> pq;
    for (int c : count) {
        if (c > 0) pq.push(c);
    }
    
    int time = 0;
    while (!pq.empty()) {
        vector<int> temp;
        
        for (int i = 0; i <= n; i++) {
            if (!pq.empty()) {
                temp.push_back(pq.top() - 1);
                pq.pop();
            }
        }
        
        for (int t : temp) {
            if (t > 0) pq.push(t);
        }
        
        time += pq.empty() ? temp.size() : n + 1;
    }
    
    return time;
}

int main() {
    vector<char> tasks = {'A', 'A', 'A', 'B', 'B', 'B'};
    int n = 2;
    
    cout << leastInterval(tasks, n) << endl;  // 8
    
    return 0;
}
```

## 练习题

### 题目1：用队列实现栈
**题目描述**：使用两个队列实现栈。

**参考代码**：
```cpp
#include <iostream>
#include <queue>
using namespace std;

class MyStack {
private:
    queue<int> q1, q2;
    
public:
    void push(int x) {
        q2.push(x);
        while (!q1.empty()) {
            q2.push(q1.front());
            q1.pop();
        }
        swap(q1, q2);
    }
    
    int pop() {
        int val = q1.front();
        q1.pop();
        return val;
    }
    
    int top() {
        return q1.front();
    }
    
    bool empty() {
        return q1.empty();
    }
};

int main() {
    MyStack st;
    st.push(1);
    st.push(2);
    st.push(3);
    
    cout << st.top() << endl;   // 3
    cout << st.pop() << endl;   // 3
    cout << st.pop() << endl;   // 2
    cout << st.empty() << endl; // 0
    
    return 0;
}
```

### 题目2：二叉树层序遍历
**题目描述**：给定一个二叉树，返回其节点值的层序遍历（逐层从左到右）。

**参考代码**：
```cpp
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

struct TreeNode {
    int val;
    TreeNode* left;
    TreeNode* right;
    TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
};

vector<vector<int>> levelOrder(TreeNode* root) {
    vector<vector<int>> result;
    if (!root) return result;
    
    queue<TreeNode*> q;
    q.push(root);
    
    while (!q.empty()) {
        int levelSize = q.size();
        vector<int> level;
        
        for (int i = 0; i < levelSize; i++) {
            TreeNode* node = q.front();
            q.pop();
            level.push_back(node->val);
            
            if (node->left) q.push(node->left);
            if (node->right) q.push(node->right);
        }
        
        result.push_back(level);
    }
    
    return result;
}

int main() {
    TreeNode* root = new TreeNode(3);
    root->left = new TreeNode(9);
    root->right = new TreeNode(20);
    root->right->left = new TreeNode(15);
    root->right->right = new TreeNode(7);
    
    vector<vector<int>> result = levelOrder(root);
    for (auto& level : result) {
        for (int val : level) cout << val << " ";
        cout << endl;
    }
    
    return 0;
}
```

## 小结

1. 队列是FIFO结构，适合按顺序处理
2. BFS天然使用队列实现
3. 循环队列可避免空间浪费
4. 优先队列（堆）用于维护最大/最小值
