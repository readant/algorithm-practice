# C 常用函数

## 一、数学函数

```c
#include <math.h>

// 绝对值
abs(-5);                   // 5
fabs(-3.14);               // 3.14

// 幂运算
pow(2, 10);                // 1024.0

// 平方根
sqrt(16);                  // 4.0

// 向下取整
floor(3.7);                // 3.0

// 向上取整
ceil(3.2);                 // 4.0

// 四舍五入
round(3.6);                // 4.0
round(3.4);                // 3.0

// 三角函数
sin(M_PI / 2);             // 1.0
cos(0);                    // 1.0
tan(M_PI / 4);             // 1.0

// 对数
log(10);                   // 自然对数
log10(10);                 // 以10为底
```

## 二、字符串函数

```c
#include <string.h>

// 字符串长度
strlen("hello");           // 5

// 字符串复制
char dest[20];
strcpy(dest, "hello");     // 复制

// 字符串连接
strcat(dest, " world");    // 连接

// 字符串比较
strcmp("abc", "abc");      // 0（相等）
strcmp("abc", "abd");      // -1（小于）
strcmp("abd", "abc");      // 1（大于）

// 字符串查找
strstr("hello world", "world");  // 查找子串
strchr("hello", 'l');             // 查找字符

// 字符串分割
char str[] = "hello,world,python";
char *token = strtok(str, ",");
while (token != NULL) {
    printf("%s\n", token);
    token = strtok(NULL, ",");
}

// 内存操作
memset(arr, 0, sizeof(arr));     // 清零
memcpy(dest, src, n);             // 复制内存
memcmp(arr1, arr2, n);            // 比较内存
```

## 三、字符函数

```c
#include <ctype.h>

// 判断字符
isdigit('5');             // 1（数字）
isalpha('a');             // 1（字母）
islower('a');             // 1（小写）
isupper('A');             // 1（大写）
isalnum('a');             // 1（字母或数字）
isspace(' ');             // 1（空白字符）

// 转换字符
tolower('A');             // 'a'
toupper('a');             // 'A'
```

## 四、随机数

```c
#include <stdlib.h>
#include <time.h>

// 设置随机种子
srand(time(NULL));

// 生成随机数
int rand_num = rand();           // 0 到 RAND_MAX
int rand_num = rand() % 100;     // 0 到 99
int rand_num = rand() % 100 + 1; // 1 到 100

// 生成指定范围随机数
int random(int min, int max) {
    return rand() % (max - min + 1) + min;
}
```

## 五、排序和查找

```c
#include <stdlib.h>

// 比较函数
int compare(const void *a, const void *b) {
    return (*(int*)a - *(int*)b);
}

// 排序
int arr[] = {3, 1, 4, 1, 5};
int n = sizeof(arr) / sizeof(arr[0]);
qsort(arr, n, sizeof(int), compare);

// 查找
int target = 4;
int *found = bsearch(&target, arr, n, sizeof(int), compare);
if (found != NULL) {
    printf("找到: %d\n", *found);
}
```

## 六、其他常用函数

```c
#include <stdlib.h>

// 动态内存分配
void *malloc(size_t size);       // 分配内存
void *calloc(size_t num, size_t size);  // 分配并清零
void *realloc(void *ptr, size_t size);  // 重新分配
void free(void *ptr);            // 释放内存

// 程序控制
exit(0);                         // 正常退出
exit(1);                         // 异常退出
system("pause");                 // 执行系统命令

// 类型转换
atoi("123");                     // 字符串转整数
atof("3.14");                    // 字符串转浮点数
itoa(123, str, 10);              // 整数转字符串（非标准）

// 时间函数
#include <time.h>
time_t now = time(NULL);         // 当前时间
printf("%s", ctime(&now));       // 格式化输出
```
