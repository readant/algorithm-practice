# Python 常用函数

## 一、数学函数

```python
import math

# 基础运算
abs(-5)                 # 5
max(1, 2, 3)            # 3
min(1, 2, 3)            # 1
sum([1, 2, 3])          # 6

# 幂运算
pow(2, 10)              # 1024
2 ** 10                 # 1024

# 平方根
math.sqrt(16)           # 4.0

# 向下取整
math.floor(3.7)         # 3

# 向上取整
math.ceil(3.2)          # 4

# 四舍五入
round(3.6)              # 4
round(3.4)              # 3
round(3.5)              # 4

# 最大公约数
math.gcd(12, 18)        # 6

# 最小公倍数
math.lcm(12, 18)        # 36（Python 3.9+）

# 对数
math.log(10)            # 自然对数
math.log10(10)          # 以10为底
math.log2(8)            # 以2为底

# 三角函数
math.sin(math.pi/2)     # 1.0
math.cos(0)             # 1.0
math.tan(math.pi/4)     # 1.0
```

## 二、排序函数

```python
# 列表排序
lst = [3, 1, 4, 1, 5]
lst.sort()              # 就地排序 [1, 1, 3, 4, 5]
lst.sort(reverse=True)  # 降序 [5, 4, 3, 1, 1]

# 返回新列表
sorted_lst = sorted(lst)

# 自定义排序
lst.sort(key=lambda x: abs(x))      # 按绝对值
lst.sort(key=lambda x: -x)          # 降序
lst.sort(key=lambda x: x % 2)       # 按奇偶

# 多条件排序
lst = [(1, 3), (2, 2), (1, 2)]
lst.sort(key=lambda x: (x[0], x[1]))  # 先按第一个元素，再按第二个
```

## 三、查找函数

```python
# 列表查找
lst = [1, 2, 3, 4, 5]
lst.index(3)            # 2（位置）
3 in lst                # True
lst.count(3)            # 1（出现次数）

# 字符串查找
s = "hello world"
s.find("world")         # 6
s.find("python")        # -1
"world" in s            # True

# 字典查找
d = {'a': 1, 'b': 2}
'a' in d                # True
d.get('a', 0)           # 1
d.get('c', 0)           # 0（默认值）
```

## 四、类型转换

```python
# 数字转字符串
str(123)                # "123"
str(3.14)               # "3.14"

# 字符串转数字
int("123")              # 123
float("3.14")           # 3.14

# 进制转换
bin(10)                 # "0b1010"
oct(10)                 # "0o12"
hex(10)                 # "0xa"
int("1010", 2)          # 10

# 列表转字符串
",".join(["a", "b", "c"])  # "a,b,c"

# 字符串转列表
list("hello")           # ['h', 'e', 'l', 'l', 'o']
"hello".split()         # ['hello']
"a,b,c".split(",")      # ['a', 'b', 'c']
```

## 五、其他常用函数

```python
# 长度
len([1, 2, 3])          # 3
len("hello")            # 5

# 范围
list(range(5))          # [0, 1, 2, 3, 4]
list(range(1, 6))       # [1, 2, 3, 4, 5]
list(range(0, 10, 2))   # [0, 2, 4, 6, 8]

# 枚举
for i, x in enumerate([1, 2, 3]):
    print(i, x)         # 0 1, 1 2, 2 3

# 拉链
a = [1, 2, 3]
b = ['a', 'b', 'c']
list(zip(a, b))         # [(1, 'a'), (2, 'b'), (3, 'c')]

# 过滤
list(filter(lambda x: x > 2, [1, 2, 3, 4]))  # [3, 4]

# 映射
list(map(lambda x: x * 2, [1, 2, 3]))       # [2, 4, 6]

# 反转
list(reversed([1, 2, 3]))                   # [3, 2, 1]
[1, 2, 3][::-1]                             # [3, 2, 1]
```
