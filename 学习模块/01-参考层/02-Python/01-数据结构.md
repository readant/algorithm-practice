# Python 数据结构

## 一、列表

```python
# 创建
lst = [1, 2, 3, 4, 5]
lst = [0] * 5           # [0, 0, 0, 0, 0]
lst = list(range(5))    # [0, 1, 2, 3, 4]

# 常用操作
lst.append(6)           # 末尾添加
lst.insert(0, 0)        # 指定位置插入
lst.pop()               # 末尾删除
lst.pop(0)              # 指定位置删除
lst.remove(3)           # 删除指定值
lst.index(3)            # 查找位置
lst.count(3)            # 计数

# 切片
lst[1:4]                # [2, 3, 4]
lst[::-1]               # 反转 [5, 4, 3, 2, 1]
lst[::2]                # 隔一个取一个

# 排序
lst.sort()              # 就地排序
lst.sort(reverse=True)  # 降序
sorted(lst)             # 返回新列表

# 列表推导式
lst = [i * 2 for i in range(5)]  # [0, 2, 4, 6, 8]
lst = [x for x in lst if x > 2]  # 过滤
```

## 二、元组

```python
# 创建（不可变）
tup = (1, 2, 3)
tup = 1, 2, 3

# 访问
tup[0]                  # 1
tup[1:3]                # (2, 3)

# 解包
a, b, c = tup
```

## 三、字典

```python
# 创建
d = {'a': 1, 'b': 2}
d = dict(a=1, b=2)
d = {i: i * 2 for i in range(3)}  # {0: 0, 1: 2, 2: 4}

# 访问
d['a']                  # 1
d.get('c', 0)           # 0（默认值）
d.keys()                # 获取所有键
d.values()              # 获取所有值
d.items()               # 获取所有键值对

# 操作
d['c'] = 3              # 添加/修改
d.pop('a')              # 删除
d.update({'d': 4})      # 批量更新

# 遍历
for k, v in d.items():
    print(k, v)
```

## 四、集合

```python
# 创建
s = {1, 2, 3}
s = set([1, 2, 2, 3])   # {1, 2, 3}（自动去重）

# 操作
s.add(4)                # 添加
s.remove(1)             # 删除（不存在会报错）
s.discard(1)            # 删除（不存在不会报错）

# 集合运算
a = {1, 2, 3}
b = {3, 4, 5}
a | b                   # 并集 {1, 2, 3, 4, 5}
a & b                   # 交集 {3}
a - b                   # 差集 {1, 2}
a ^ b                   # 对称差 {1, 2, 4, 5}
```

## 五、堆

```python
import heapq

# 创建小顶堆
heap = [3, 1, 4, 1, 5]
heapq.heapify(heap)

# 操作
heapq.heappush(heap, 2)  # 添加
heapq.heappop(heap)      # 弹出最小值
heap[0]                  # 查看最小值

# 大顶堆（用负数）
heap = [-x for x in [3, 1, 4, 1, 5]]
heapq.heapify(heap)
max_val = -heapq.heappop(heap)
```
