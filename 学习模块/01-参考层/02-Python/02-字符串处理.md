# Python 字符串处理

## 一、基本操作

```python
s = "hello world"

# 长度
len(s)                  # 11

# 大小写
s.upper()               # "HELLO WORLD"
s.lower()               # "hello world"
s.capitalize()          # "Hello world"
s.title()               # "Hello World"

# 查找
s.find("world")         # 6
s.find("python")        # -1（未找到）
s.index("world")        # 6（未找到会报错）
s.count("l")            # 3

# 判断
s.startswith("hello")   # True
s.endswith("world")     # True
s.isalpha()             # False（包含空格）
s.isdigit()             # False
s.isalnum()             # False
```

## 二、切片和替换

```python
s = "hello world"

# 切片
s[0:5]                  # "hello"
s[6:]                   # "world"
s[::-1]                 # "dlrow olleh"

# 替换
s.replace("world", "python")  # "hello python"
s.replace("l", "x", 1)        # "hexlo world"（只替换1次）

# 删除
s.strip()               # 删除首尾空白
s.lstrip()              # 删除首部空白
s.rstrip()              # 删除尾部空白
s.strip("h")            # 删除首尾的'h'
```

## 三、分割和连接

```python
s = "hello world python"

# 分割
s.split()               # ["hello", "world", "python"]
s.split(" ", 1)         # ["hello", "world python"]（最多分割1次）
s.splitlines()          # 按行分割

# 连接
" ".join(["hello", "world"])  # "hello world"
"-".join(["a", "b", "c"])     # "a-b-c"
```

## 四、格式化

```python
# f-string（Python 3.6+）
name = "Tom"
age = 18
f"姓名: {name}, 年龄: {age}"  # "姓名: Tom, 年龄: 18"

# 格式化
f"{pi:.2f}"             # 保留2位小数
f"{num:05d}"            # 补零到5位
f"{x:>10}"              # 右对齐，宽度10
f"{x:<10}"              # 左对齐，宽度10
f"{x:^10}"              # 居中，宽度10

# format()
"姓名: {}, 年龄: {}".format(name, age)
"姓名: {0}, 年龄: {1}".format(name, age)
"姓名: {n}, 年龄: {a}".format(n=name, a=age)

# %
"姓名: %s, 年龄: %d" % (name, age)
```

## 五、类型转换

```python
# 数字转字符串
str(123)                # "123"
str(3.14)               # "3.14"

# 字符串转数字
int("123")              # 123
float("3.14")           # 3.14

# 进制转换
bin(10)                 # "0b1010"
oct(10)                 # "0o12"
hex(10)                 # "0xa"
int("1010", 2)          # 10（二进制转十进制）
```

## 六、常用技巧

```python
# 反转字符串
s[::-1]

# 检查回文
s == s[::-1]

# 统计字符频率
from collections import Counter
Counter(s)

# 去除重复字符
"".join(set(s))

# 判断是否为数字
s.isdigit()

# 判断是否为字母
s.isalpha()
```
